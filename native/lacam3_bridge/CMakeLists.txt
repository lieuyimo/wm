cmake_minimum_required(VERSION 3.16)
project(wm_lacam3_bridge LANGUAGES CXX)
if(NOT DEFINED LACAM3_UPSTREAM)
  message(FATAL_ERROR "Set LACAM3_UPSTREAM to the pinned official checkout")
endif()
add_subdirectory("${LACAM3_UPSTREAM}/lacam3" official-lacam3)
add_executable(wm_lacam3_worker lacam3_worker.cpp)
target_compile_features(wm_lacam3_worker PRIVATE cxx_std_17)
target_include_directories(wm_lacam3_worker PRIVATE "${LACAM3_UPSTREAM}/lacam3/include")
target_link_libraries(wm_lacam3_worker PRIVATE lacam3)
